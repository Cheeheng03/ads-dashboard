<script>
  (() => {
    function initPupsSlider(scope) {
      const section = scope.querySelector('[data-pups-section]') || scope;

      if (!section || section.dataset.pupsInitialized === 'true') return;

      const viewport = section.querySelector('[data-slider-viewport]');
      const track = section.querySelector('[data-slider-track]');
      const slides = Array.from(section.querySelectorAll('[data-slide]'));
      const prevBtn = section.querySelector('[data-prev]');
      const nextBtn = section.querySelector('[data-next]');
      const dotsWrap = section.querySelector('[data-dots]');
      const controls = section.querySelector('[data-pups-controls]');

      if (!viewport || !track || !slides.length || !prevBtn || !nextBtn || !dotsWrap) {
        if (controls) controls.hidden = true;
        return;
      }

      section.dataset.pupsInitialized = 'true';

      let currentIndex = 0;
      let currentSlidesPerView = 4;
      let pages = 1;

      function getSlidesPerView() {
        if (window.innerWidth <= 749) return 1;
        if (window.innerWidth <= 1199) return 2;
        return 4;
      }

      function buildDots() {
        dotsWrap.innerHTML = '';

        for (let i = 0; i < pages; i++) {
          const dot = document.createElement('button');
          dot.type = 'button';
          dot.className = 'pups__dot' + (i === currentIndex ? ' is-active' : '');
          dot.setAttribute('aria-label', 'Go to slide group ' + (i + 1));

          dot.addEventListener('click', () => {
            currentIndex = i;
            update();
          });

          dotsWrap.appendChild(dot);
        }
      }

      function updateButtons() {
        prevBtn.disabled = currentIndex <= 0;
        nextBtn.disabled = currentIndex >= pages - 1;

        if (controls) {
          controls.hidden = pages <= 1;
        }
      }

      function update() {
        currentSlidesPerView = getSlidesPerView();
        pages = Math.max(1, Math.ceil(slides.length / currentSlidesPerView));

        if (currentIndex > pages - 1) {
          currentIndex = pages - 1;
        }

        const firstSlide = slides[0];
        const trackStyles = getComputedStyle(track);
        const gap = parseFloat(trackStyles.columnGap || trackStyles.gap || 0);
        const slideWidth = firstSlide.getBoundingClientRect().width;
        const offset = currentIndex * ((slideWidth + gap) * currentSlidesPerView);

        track.style.transform = `translate3d(${-offset}px, 0, 0)`;

        buildDots();
        updateButtons();
      }

      prevBtn.addEventListener('click', () => {
        if (currentIndex > 0) {
          currentIndex -= 1;
          update();
        }
      });

      nextBtn.addEventListener('click', () => {
        if (currentIndex < pages - 1) {
          currentIndex += 1;
          update();
        }
      });

      let resizeTimer;

      window.addEventListener('resize', () => {
        clearTimeout(resizeTimer);
        resizeTimer = setTimeout(update, 120);
      });

      update();
    }

    if (!customElements.get('product-recommendations')) {
      customElements.define(
        'product-recommendations',
        class ProductRecommendations extends HTMLElement {
          connectedCallback() {
            if (this.dataset.loaded === 'true') return;

            const url = this.dataset.url;

            if (!url) return;

            const loadRecommendations = () => {
              fetch(url)
                .then((response) => response.text())
                .then((text) => {
                  const html = new DOMParser().parseFromString(text, 'text/html');
                  const recommendations = html.querySelector('product-recommendations');

                  if (recommendations && recommendations.innerHTML.trim().length) {
                    this.innerHTML = recommendations.innerHTML;
                    this.dataset.loaded = 'true';
                    initPupsSlider(this);
                  } else {
                    this.remove();
                  }
                })
                .catch((error) => {
                  console.error('Product recommendations failed:', error);
                  this.remove();
                });
            };

            if ('IntersectionObserver' in window) {
              const observer = new IntersectionObserver((entries, observerInstance) => {
                if (!entries[0].isIntersecting) return;

                observerInstance.unobserve(this);
                loadRecommendations();
              }, {
                rootMargin: '0px 0px 300px 0px'
              });

              observer.observe(this);
            } else {
              loadRecommendations();
            }
          }
        }
      );
    }

    document.querySelectorAll('product-recommendations').forEach((element) => {
      if (element.dataset.loaded === 'true' || element.querySelector('[data-pups-section]')) {
        initPupsSlider(element);
      }
    });

    document.addEventListener('shopify:section:load', (event) => {
      const recommendations = event.target.querySelector('product-recommendations');

      if (recommendations) {
        recommendations.dataset.loaded = '';
      }

      document.querySelectorAll('product-recommendations').forEach((element) => {
        if (element.querySelector('[data-pups-section]')) {
          initPupsSlider(element);
        }
      });
    });
  })();
</script>
